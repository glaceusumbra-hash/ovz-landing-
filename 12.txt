<!-- SCRIPT SIMPLE DE TABS -->
<script>
const tabs = document.querySelectorAll('.tab-label');
const views = document.querySelectorAll('.view-panel');
const texts = document.querySelectorAll('.text-content');

tabs.forEach(tab => {
    tab.addEventListener('click', () => {
        // Limpiar activos
        tabs.forEach(t => t.classList.remove('active'));
        views.forEach(v => v.classList.remove('active'));
        texts.forEach(txt => txt.classList.remove('active'));
        
        // Asignar nuevo activo
        tab.classList.add('active');
        let targetId = tab.getAttribute('data-target');
        
        document.getElementById('view-' + targetId).classList.add('active');
        document.getElementById('text-' + targetId).classList.add('active');
    });
});
</script>
